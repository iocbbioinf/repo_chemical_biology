{#- Helpers for the dataset record detail page. Field labels come from the model (metadata.yaml), enum values are shown via the `labels` dictionary when given. -#} {%- macro detail_value(fd, labels=None) -%} {%- set raw = fd.value(fd) -%} {{- labels.get(raw, raw) if labels else fd.ui_value(fd) -}} {%- endmacro -%} {#- One table row; nothing is rendered when the field is empty. -#} {%- macro detail_row(fd, labels=None, multiple=False) -%} {%- if fd.value(fd) -%} {{ fd.label(fd, "") }} {%- if multiple -%} {#- raw API values; FieldData.array() needs UI serialized list data -#} {%- for raw in fd.value(fd) -%} {{- ", " if not loop.first }}{{ labels.get(raw, raw) if labels else raw -}} {%- endfor -%} {%- else -%} {{- detail_value(fd, labels) -}} {%- endif -%} {%- endif -%} {%- endmacro -%} {#- English title of a vocabulary item, falls back to its id. The UI serialization of custom vocabulary fields is empty, so the stored (API) title is used. -#} {%- macro vocab_title(fd) -%} {%- set title = fd.title.value(fd.title) -%} {{- title.get("en") if title is mapping and title.get("en") else fd.id.value(fd.id) -}} {%- endmacro -%} {#- One table row with a vocabulary item or an array of them. -#} {%- macro vocab_row(fd, multiple=False) -%} {%- if fd.value(fd) -%} {{ fd.label(fd, "") }} {%- if multiple -%} {%- for item in fd.array(fd) -%} {{- ", " if not loop.first }}{{ vocab_title(item) -}} {%- endfor -%} {%- else -%} {{- vocab_title(fd) -}} {%- endif -%} {%- endif -%} {%- endmacro -%} {#- Row with contributors of the given role (principal investigator = ProjectLeader, sample preparer = DataCollector) and the e-mail kept next to them. -#} {%- macro contact_row(label, contributors, role, email_fd) -%} {%- set people = contributors | selectattr("role") | selectattr("role.id", "equalto", role) | list -%} {%- set email = email_fd.value(email_fd) -%} {%- if people or email -%} {{ label }} {%- for c in people %} {{- "
" | safe if not loop.first }}{{ c.person_or_org.name }} {%- for aff in c.affiliations or [] %} {{- " (" if loop.first else ", " -}} {%- if aff.id %}{{ aff.name or aff.id }}{% else %}{{ aff.name }}{% endif -%} {{- ")" if loop.last -}} {%- endfor %} {%- endfor %} {%- if email %}{{ "
" | safe if people }}{{ email }}{% endif %} {%- endif -%} {%- endmacro -%} {#- Row with geolocations (metadata.locations.features): place, description and point coordinates linked to OpenStreetMap. -#} {%- macro locations_row(fd, label) -%} {%- set features = (fd.value(fd) or {}).get("features") or [] -%} {%- if features -%} {{ label }} {%- for f in features %} {%- set coords = (f.geometry or {}).get("coordinates") if (f.geometry or {}).get("type") == "Point" else None -%} {{- "
" | safe if not loop.first }}{{ f.place or "" }} {%- if f.description %}{{ " – " if f.place }}{{ f.description }}{% endif %} {%- if coords %} {%- set lat, lon = coords[1], coords[0] -%} {{ " " if f.place or f.description }}{{ lat }}, {{ lon }} {%- endif %} {%- endfor %} {%- endif -%} {%- endmacro -%}
